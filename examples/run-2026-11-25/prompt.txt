+import { db } from "../db.ts";
+import { gateway } from "./gateway.ts";
+
+export type RefundRequest = {
+  orderId: string;
+  amount: number; // dollars
+  reason?: string;
+};
+
+/** Refunds part or all of an order. Called by the admin page and by the gateway webhook. */
+export async function refundOrder(request: RefundRequest): Promise<{ ok: boolean }> {
+  const order = await db.order.findUnique({ where: { id: request.orderId } });
+  if (!order) return { ok: false };
+
+  const alreadyRefunded = order.refundedCents / 100;
+  if (request.amount > order.paidCents / 100 - alreadyRefunded) {
+    return { ok: false };
+  }
+
+  try {
+    await gateway.refund({
+      chargeId: order.chargeId,
+      amountCents: request.amount * 100,
+    });
+  } catch (error) {
+    console.error("refund failed", error);
+  }
+
+  await db.order.update({
+    where: { id: order.id },
+    data: { refundedCents: order.refundedCents + request.amount * 100 },
+  });
+
+  await db.auditLog.create({
+    data: { orderId: order.id, action: "refund", note: request.reason ?? "" },
+  });
+  return { ok: true };
+}
